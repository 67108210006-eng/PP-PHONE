async function logout() { if (!confirm('คุณต้องการออกจากระบบใช่หรือไม่?')) { return; } try { // 1. เคลียร์ Session จาก Supabase (ถ้ามี) if (typeof DB !== 'undefined' && DB.getClient) { const client = DB.getClient(); if (client && client.auth) { await client.auth.signOut(); } } } catch (err) { console.warn('Supabase logout error:', err); } // 2. เคลียร์ LocalStorage และ SessionStorage ทั้งหมด localStorage.removeItem('currentUser'); localStorage.removeItem('pp_user_role'); localStorage.clear(); sessionStorage.clear(); // 3. เปลี่ยนหน้าไปยัง login.html window.location.href = 'login.html'; }
